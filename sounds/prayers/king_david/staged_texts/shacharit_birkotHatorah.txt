בָּרוּךְ אַתָּה אָדוֹנַי אֱלֹהֵֽינוּ מֶֽלֶךְ הָעוֹלָם אֲשֶׁר קִדְּ֒שָֽׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לַעֲסֹק בְּדִבְרֵי תוֹרָה וְהַעֲרֶב נָא אָדוֹנַי אֱלֹהֵֽינוּ אֶת־דִּבְרֵי תוֹרָתְךָ בְּפִֽינוּ וּבְפִי עַמְּךָ בֵּית יִשְׂרָאֵל. וְנִהְיֶה אֲנַֽחְנוּ וְצֶאֱצָאֵֽינוּ וְצֶאֱצָאֵי עַמְּךָ בֵּית יִשְׂרָאֵל כֻּלָּֽנוּ יוֹדְעֵי שְׁמֶֽךָ וְלוֹמְדֵי תוֹרָתֶֽךָ. (י"א תוֹרָתְךָ) לִשְׁמָהּ. בָּרוּךְ אַתָּה אָדוֹנַי הַמְלַמֵּד תּוֹרָה לְעַמּוֹ יִשְׂרָאֵל. בָּרוּךְ אַתָּה אָדוֹנַי אֱלֹהֵֽינוּ מֶֽלֶךְ הָעוֹלָם אֲשֶׁר בָּֽחַר בָּֽנוּ מִכָּל הָעַמִּים וְנָֽתַן לָֽנוּ אֶת־תּוֹרָתוֹ בָּרוּךְ אַתָּה אָדוֹנַי נוֹתֵן הַתּוֹרָה יְבָרֶכְ֒ךָ אָדוֹנַי וְיִשְׁמְרֶֽךָ יָאֵר אָדוֹנַי פָּנָיו אֵלֶֽיךָ וִיחֻנֶּֽךָּ יִשָּׂא אָדוֹנַי פָּנָיו אֵלֶֽיךָ וְיָשֵׂם לְךָ שָׁלוֹם אֵֽלּוּ דְבָרִים שֶׁאֵין לָהֶם שִׁעוּר הַפֵּאָה וְהַבִּכּוּרִים וְהָרְ֒אָיוֹן וּגְמִילוּת חֲסָדִים וְתַלְמוּד תּוֹרָה אֵֽלּוּ דְבָרִים שֶׁאָדָם אוֹכֵל פֵּרוֹתֵיהֶם בָּעוֹלָם הַזֶּה וְהַקֶּֽרֶן קַיֶּֽמֶת לָעוֹלָם הַבָּא, וְאֵֽלּוּ הֵן כִּבּוּד אָב וָאֵם וּגְמִילוּת חֲסָדִים וְהַשְׁכָּמַת בֵּית הַמִּדְרָשׁ שַׁחֲרִית וְעַרְבִית וְהַכְנָסַת אוֹרְ֒חִים וּבִקּוּר חוֹלִים וְהַכְנָסַת כַּלָּה וּלְוָיַת הַמֵּת וְעִיּוּן תְּפִלָּה וַהֲבָאַת שָׁלוֹם בֵּין אָדָם לַחֲבֵרוֹ ובֵין אִישׁ לְאִשְׁתּוֹ וְתַלְמוּד תּוֹרָה כְּנֶֽגֶד כֻּלָּם
